{{- /* Drop-in for `markdownify` that renders in the context of the owning page, so render hooks (render-link) resolve relative links against that page rather than the home page. Like markdownify, a lone paragraph is unwrapped. Usage: {{ partial "utils/markdownify.html" (dict "page" .Page "text" .Inner) }} */ -}} {{- $html := .page.RenderString (dict "display" "block") .text | string | strings.TrimSpace -}} {{- if and (strings.HasPrefix $html "

") (strings.HasSuffix $html "

") (eq (len (findRE `]` $html)) 1) -}} {{- $html = $html | strings.TrimPrefix "

" | strings.TrimSuffix "

" -}} {{- end -}} {{- return ($html | safeHTML) -}}
